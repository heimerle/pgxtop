67    pub(crate) fn refresh(&mut self) -> bool {
68        unsafe {
69            if let Some(requested_properties) = build_requested_properties(&[
70                ffi::kCFURLVolumeAvailableCapacityKey,
71                ffi::kCFURLVolumeAvailableCapacityForImportantUsageKey,
72            ]) {
73                match get_disk_properties(&self.volume_url, &requested_properties) {
74                    Some(disk_props) => {
75                        self.available_space = get_available_volume_space(&disk_props);
76                        true
77                    }
78                    None => {
79                        sysinfo_debug!("Failed to get disk properties");
80                        false
81                    }
82                }
83            } else {
84                sysinfo_debug!("failed to create volume key list, skipping refresh");
85                false
86            }
87        }
88    }
89}

117unsafe fn get_list(container: &mut Vec<Disk>) {
118    container.clear();
119
120    let raw_disks = {
121        let count = libc::getfsstat(ptr::null_mut(), 0, libc::MNT_NOWAIT);
122        if count < 1 {
123            return;
124        }
125        let bufsize = count * std::mem::size_of::<libc::statfs>() as libc::c_int;
126        let mut disks = Vec::with_capacity(count as _);
127        let count = libc::getfsstat(disks.as_mut_ptr(), bufsize, libc::MNT_NOWAIT);
128
129        if count < 1 {
130            return;
131        }
132
133        disks.set_len(count as usize);
134
135        disks
136    };
137
138    // Create a list of properties about the disk that we want to fetch.
139    let requested_properties = match build_requested_properties(&[
140        ffi::kCFURLVolumeIsEjectableKey,
141        ffi::kCFURLVolumeIsRemovableKey,
142        ffi::kCFURLVolumeIsInternalKey,
143        ffi::kCFURLVolumeTotalCapacityKey,
144        ffi::kCFURLVolumeAvailableCapacityForImportantUsageKey,
145        ffi::kCFURLVolumeAvailableCapacityKey,
146        ffi::kCFURLVolumeNameKey,
147        ffi::kCFURLVolumeIsBrowsableKey,
148        ffi::kCFURLVolumeIsLocalKey,
149    ]) {
150        Some(properties) => properties,
151        None => {
152            sysinfo_debug!("failed to create volume key list");
153            return;
154        }
155    };
156
157    for c_disk in raw_disks {
158        let volume_url = match CFReleaser::new(
159            core_foundation_sys::url::CFURLCreateFromFileSystemRepresentation(
160                kCFAllocatorDefault,
161                c_disk.f_mntonname.as_ptr() as *const _,
162                c_disk.f_mntonname.len() as _,
163                false as _,
164            ),
165        ) {
166            Some(url) => url,
167            None => {
168                sysinfo_debug!("getfsstat returned incompatible paths");
169                continue;
170            }
171        };
172
173        let prop_dict = match get_disk_properties(&volume_url, &requested_properties) {
174            Some(props) => props,
175            None => continue,
176        };
177
178        // Future note: There is a difference between `kCFURLVolumeIsBrowsableKey` and the
179        // `kCFURLEnumeratorSkipInvisibles` option of `CFURLEnumeratorOptions`. Specifically,
180        // the first one considers the writable `Data`(`/System/Volumes/Data`) partition to be
181        // browsable, while it is classified as "invisible" by CoreFoundation's volume emumerator.
182        let browsable = get_bool_value(
183            prop_dict.inner(),
184            DictKey::Extern(ffi::kCFURLVolumeIsBrowsableKey),
185        )
186        .unwrap_or_default();
187
188        // Do not return invisible "disks". Most of the time, these are APFS snapshots, hidden
189        // system volumes, etc. Browsable is defined to be visible in the system's UI like Finder,
190        // disk utility, system information, etc.
191        //
192        // To avoid seemingly duplicating many disks and creating an inaccurate view of the system's
193        // resources, these are skipped entirely.
194        if !browsable {
195            continue;
196        }
197
198        let local_only = get_bool_value(
199            prop_dict.inner(),
200            DictKey::Extern(ffi::kCFURLVolumeIsLocalKey),
201        )
202        .unwrap_or(true);
203
204        // Skip any drive that is not locally attached to the system.
205        //
206        // This includes items like SMB mounts, and matches the other platform's behavior.
207        if !local_only {
208            continue;
209        }
210
211        let mount_point = PathBuf::from(OsStr::from_bytes(
212            CStr::from_ptr(c_disk.f_mntonname.as_ptr()).to_bytes(),
213        ));
214
215        if let Some(disk) = new_disk(mount_point, volume_url, c_disk, &prop_dict) {
216            container.push(disk);
217        }
218    }
219}

221type RetainedCFArray = CFReleaser<core_foundation_sys::array::__CFArray>;
222type RetainedCFDictionary = CFReleaser<core_foundation_sys::dictionary::__CFDictionary>;
223type RetainedCFURL = CFReleaser<core_foundation_sys::url::__CFURL>;
224
225unsafe fn build_requested_properties(properties: &[CFStringRef]) -> Option<RetainedCFArray> {
226    CFReleaser::new(CFArrayCreate(
227        ptr::null_mut(),
228        properties.as_ptr() as *const *const c_void,
229        properties.len() as _,
230        &core_foundation_sys::array::kCFTypeArrayCallBacks,
231    ))
232}
233
234fn get_disk_properties(
235    volume_url: &RetainedCFURL,
236    requested_properties: &RetainedCFArray,
237) -> Option<RetainedCFDictionary> {
238    CFReleaser::new(unsafe {
239        ffi::CFURLCopyResourcePropertiesForKeys(
240            volume_url.inner(),
241            requested_properties.inner(),
242            ptr::null_mut(),
243        )
244    })
245}

247fn get_available_volume_space(disk_props: &RetainedCFDictionary) -> u64 {
248    // We prefer `AvailableCapacityForImportantUsage` over `AvailableCapacity` because
249    // it takes more of the system's properties into account, like the trash, system-managed caches,
250    // etc. It generally also returns higher values too, because of the above, so it's a more
251    // accurate representation of what the system _could_ still use.
252    unsafe {
253        get_int_value(
254            disk_props.inner(),
255            DictKey::Extern(ffi::kCFURLVolumeAvailableCapacityForImportantUsageKey),
256        )
257        .filter(|bytes| *bytes != 0)
258        .or_else(|| {
259            get_int_value(
260                disk_props.inner(),
261                DictKey::Extern(ffi::kCFURLVolumeAvailableCapacityKey),
262            )
263        })
264    }
265    .unwrap_or_default() as u64
266}

268pub(super) enum DictKey {
269    Extern(CFStringRef),
270    #[cfg(target_os = "macos")]
271    Defined(&'static str),
272}
273
274unsafe fn get_dict_value<T, F: FnOnce(*const c_void) -> Option<T>>(
275    dict: CFDictionaryRef,
276    key: DictKey,
277    callback: F,
278) -> Option<T> {
279    #[cfg(target_os = "macos")]
280    let _defined;
281    #[allow(clippy::infallible_destructuring_match)]
282    let key = match key {
283        DictKey::Extern(val) => val,
284        #[cfg(target_os = "macos")]
285        DictKey::Defined(val) => {
286            _defined = CFReleaser::new(cfs::CFStringCreateWithBytesNoCopy(
287                kCFAllocatorDefault,
288                val.as_ptr(),
289                val.len() as _,
290                cfs::kCFStringEncodingUTF8,
291                false as _,
292                core_foundation_sys::base::kCFAllocatorNull,
293            ))?;
294
295            _defined.inner()
296        }
297    };
298
299    let mut value = std::ptr::null();
300    if CFDictionaryGetValueIfPresent(dict, key.cast(), &mut value) != 0 {
301        callback(value)
302    } else {
303        None
304    }
305}

307pub(super) unsafe fn get_str_value(dict: CFDictionaryRef, key: DictKey) -> Option<String> {
308    get_dict_value(dict, key, |v| {
309        let v = v as cfs::CFStringRef;
310
311        let len_utf16 = cfs::CFStringGetLength(v) as usize;
312        let len_bytes = len_utf16 * 2; // Two bytes per UTF-16 codepoint.
313
314        let v_ptr = cfs::CFStringGetCStringPtr(v, cfs::kCFStringEncodingUTF8);
315        if v_ptr.is_null() {
316            // Fallback on CFStringGetString to read the underlying bytes from the CFString.
317            let mut buf = vec![0; len_bytes];
318            let success = cfs::CFStringGetCString(
319                v,
320                buf.as_mut_ptr(),
321                len_bytes as _,
322                cfs::kCFStringEncodingUTF8,
323            );
324
325            if success != 0 {
326                utils::vec_to_rust(buf)
327            } else {
328                None
329            }
330        } else {
331            crate::unix::utils::cstr_to_rust_with_size(v_ptr, Some(len_bytes))
332        }
333    })
334}

336unsafe fn get_bool_value(dict: CFDictionaryRef, key: DictKey) -> Option<bool> {
337    get_dict_value(dict, key, |v| Some(v as CFBooleanRef == kCFBooleanTrue))
338}

340unsafe fn get_int_value(dict: CFDictionaryRef, key: DictKey) -> Option<i64> {
341    get_dict_value(dict, key, |v| {
342        let mut val: i64 = 0;
343        if CFNumberGetValue(
344            v.cast(),
345            core_foundation_sys::number::kCFNumberSInt64Type,
346            &mut val as *mut i64 as *mut c_void,
347        ) {
348            Some(val)
349        } else {
350            None
351        }
352    })
353}

355unsafe fn new_disk(
356    mount_point: PathBuf,
357    volume_url: RetainedCFURL,
358    c_disk: libc::statfs,
359    disk_props: &RetainedCFDictionary,
360) -> Option<Disk> {
361    // IOKit is not available on any but the most recent (16+) iOS and iPadOS versions.
362    // Due to this, we can't query the medium type. All iOS devices use flash-based storage
363    // so we just assume the disk type is an SSD until Rust has a way to conditionally link to
364    // IOKit in more recent deployment versions.
365    #[cfg(target_os = "macos")]
366    let type_ = crate::sys::inner::disk::get_disk_type(&c_disk).unwrap_or(DiskKind::Unknown(-1));
367    #[cfg(not(target_os = "macos"))]
368    let type_ = DiskKind::SSD;
369
370    // Note: Since we requested these properties from the system, we don't expect
371    // these property retrievals to fail.
372
373    let name = get_str_value(
374        disk_props.inner(),
375        DictKey::Extern(ffi::kCFURLVolumeNameKey),
376    )
377    .map(OsString::from)?;
378
379    let is_removable = {
380        let ejectable = get_bool_value(
381            disk_props.inner(),
382            DictKey::Extern(ffi::kCFURLVolumeIsEjectableKey),
383        )
384        .unwrap_or_default();
385
386        let removable = get_bool_value(
387            disk_props.inner(),
388            DictKey::Extern(ffi::kCFURLVolumeIsRemovableKey),
389        )
390        .unwrap_or_default();
391
392        let is_removable = ejectable || removable;
393
394        if is_removable {
395            is_removable
396        } else {
397            // If neither `ejectable` or `removable` return `true`, fallback to checking
398            // if the disk is attached to the internal system.
399            let internal = get_bool_value(
400                disk_props.inner(),
401                DictKey::Extern(ffi::kCFURLVolumeIsInternalKey),
402            )
403            .unwrap_or_default();
404
405            !internal
406        }
407    };
408
409    let total_space = get_int_value(
410        disk_props.inner(),
411        DictKey::Extern(ffi::kCFURLVolumeTotalCapacityKey),
412    )? as u64;
413
414    let available_space = get_available_volume_space(disk_props);
415
416    let file_system = {
417        let len = c_disk
418            .f_fstypename
419            .iter()
420            .position(|&b| b == 0)
421            .unwrap_or(c_disk.f_fstypename.len());
422        OsString::from_vec(
423            c_disk.f_fstypename[..len]
424                .iter()
425                .map(|c| *c as u8)
426                .collect(),
427        )
428    };
429
430    let is_read_only = (c_disk.f_flags & libc::MNT_RDONLY as u32) != 0;
431
432    Some(Disk {
433        inner: DiskInner {
434            type_,
435            name,
436            file_system,
437            mount_point,
438            volume_url,
439            total_space,
440            available_space,
441            is_removable,
442            is_read_only,
443        },
444    })
445}

448/// Calls the provided closure in the context of a new autorelease pool that is drained
449/// before returning.
450/// 
451/// ## SAFETY:
452/// You must not return an Objective-C object that is autoreleased from this function since it
453/// will be freed before usable.
454unsafe fn with_autorelease<T, F: FnOnce() -> T>(call: F) -> T {
455    // NB: This struct exists to help prevent memory leaking if `call` were to panic.
456    // Otherwise, the call to `objc_autoreleasePoolPop` would never be made as the stack unwinds.
457    // `Drop` destructors for existing types on the stack are run during unwinding, so we can 
458    // ensure the autorelease pool is drained by using a RAII pattern here.
459    struct DrainPool {
460        ctx: *mut c_void,
461    }
462
463    impl Drop for DrainPool {
464        fn drop(&mut self) {
465            // SAFETY: We have not manipulated `pool_ctx` since it was received from a corresponding
466            // pool push call.
467            unsafe { ffi::objc_autoreleasePoolPop(self.ctx) }
468        }
469    }
470
471    // SAFETY: Creating a new pool is safe in any context. They can be arbitrarily nested
472    // as long as pool objects are not used in deeper layers, but we only have one and don't
473    // allow it to leave this scope.
474    let _pool_ctx = DrainPool { ctx: unsafe { ffi::objc_autoreleasePoolPush() } };
475    call()
476    // Pool is drained here before returning
477}
