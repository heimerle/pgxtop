3#[cfg(feature = "apple-sandbox")]
4pub(crate) unsafe fn get_cpu_frequency() -> u64 {
5    0
6}
7
8#[cfg(not(feature = "apple-sandbox"))]
9pub(crate) unsafe fn get_cpu_frequency() -> u64 {
10    use crate::sys::ffi;
11    use crate::sys::macos::utils::IOReleaser;
12    use crate::sys::utils::CFReleaser;
13
14    let matching = ffi::IOServiceMatching(b"AppleARMIODevice\0".as_ptr() as *const _);
15    if matching.is_null() {
16        sysinfo_debug!("IOServiceMatching call failed, `AppleARMIODevice` not found");
17        return 0;
18    }
19
20    // Starting from mac M1, the above call returns nothing for the CPU frequency
21    // so we try to get it from another source. This code comes from
22    // <https://github.com/giampaolo/psutil/pull/2222>.
23    let mut iterator: ffi::io_iterator_t = 0;
24    let result =
25        ffi::IOServiceGetMatchingServices(ffi::kIOMasterPortDefault, matching, &mut iterator);
26    if result != ffi::KIO_RETURN_SUCCESS {
27        sysinfo_debug!("Error: IOServiceGetMatchingServices() = {}", result);
28        return 0;
29    }
30    let iterator = match IOReleaser::new(iterator) {
31        Some(i) => i,
32        None => {
33            sysinfo_debug!(
34                "Error: IOServiceGetMatchingServices() succeeded but returned invalid descriptor"
35            );
36            return 0;
37        }
38    };
39
40    let mut name: ffi::io_name = std::mem::zeroed();
41    let entry = loop {
42        let entry = match IOReleaser::new(ffi::IOIteratorNext(iterator.inner())) {
43            Some(d) => d,
44            None => {
45                sysinfo_debug!("`pmgr` entry was not found in AppleARMIODevice service");
46                return 0;
47            }
48        };
49        let status = ffi::IORegistryEntryGetName(entry.inner(), name.as_mut_ptr());
50        if status != libc::KERN_SUCCESS {
51            continue;
52        } else if libc::strcmp(name.as_ptr(), b"pmgr\0".as_ptr() as *const _) == 0 {
53            break entry;
54        }
55    };
56
57    let node_name = match CFReleaser::new(ffi::CFStringCreateWithCStringNoCopy(
58        std::ptr::null(),
59        b"voltage-states5-sram\0".as_ptr() as *const _,
60        core_foundation_sys::string::kCFStringEncodingUTF8,
61        core_foundation_sys::base::kCFAllocatorNull as *mut _,
62    )) {
63        Some(n) => n,
64        None => {
65            sysinfo_debug!("CFStringCreateWithCStringNoCopy failed");
66            return 0;
67        }
68    };
69
70    let core_ref = match CFReleaser::new(ffi::IORegistryEntryCreateCFProperty(
71        entry.inner(),
72        node_name.inner(),
73        core_foundation_sys::base::kCFAllocatorDefault,
74        0,
75    )) {
76        Some(c) => c,
77        None => {
78            sysinfo_debug!("`voltage-states5-sram` property not found");
79            return 0;
80        }
81    };
82
83    let core_length = core_foundation_sys::data::CFDataGetLength(core_ref.inner() as *const _);
84    if core_length < 8 {
85        sysinfo_debug!("expected `voltage-states5-sram` buffer to have at least size 8");
86        return 0;
87    }
88    let mut max: u64 = 0;
89    core_foundation_sys::data::CFDataGetBytes(
90        core_ref.inner() as *const _,
91        core_foundation_sys::base::CFRange::init(core_length - 8, 4),
92        &mut max as *mut _ as *mut _,
93    );
94    max / 1_000_000
95}
