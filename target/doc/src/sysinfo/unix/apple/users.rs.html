11fn endswith(s1: *const c_char, s2: &[u8]) -> bool {
12    if s1.is_null() {
13        return false;
14    }
15    unsafe {
16        let mut len = strlen(s1) as isize - 1;
17        let mut i = s2.len() as isize - 1;
18        while len >= 0 && i >= 0 && *s1.offset(len) == s2[i as usize] as _ {
19            i -= 1;
20            len -= 1;
21        }
22        i == -1
23    }
24}

26pub(crate) fn get_users(users: &mut Vec<User>) {
27    fn filter(shell: *const c_char, uid: u32) -> bool {
28        !endswith(shell, b"/false") && !endswith(shell, b"/uucico") && uid < 65536
29    }
30
31    users.clear();
32
33    let mut users_map = HashMap::with_capacity(10);
34
35    unsafe {
36        setpwent();
37        loop {
38            let pw = getpwent();
39            if pw.is_null() {
40                // The call was interrupted by a signal, retrying.
41                if std::io::Error::last_os_error().kind() == std::io::ErrorKind::Interrupted {
42                    continue;
43                }
44                break;
45            }
46
47            if !filter((*pw).pw_shell, (*pw).pw_uid) {
48                // This is not a "real" or "local" user.
49                continue;
50            }
51            if let Some(name) = crate::unix::utils::cstr_to_rust((*pw).pw_name) {
52                if users_map.contains_key(&name) {
53                    continue;
54                }
55
56                let uid = (*pw).pw_uid;
57                let gid = (*pw).pw_gid;
58                users_map.insert(name, (Uid(uid), Gid(gid)));
59            }
60        }
61        endpwent();
62    }
63    for (name, (uid, gid)) in users_map {
64        users.push(User {
65            inner: UserInner::new(uid, gid, name),
66        });
67    }
68}
